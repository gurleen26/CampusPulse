--- 6. recurring_issues  (complaints collection)  -  Same category at the same location reported >= min_count times recently = recurring issue.

      {
        "$match": {
          "created_at": {

          },

          }
        }
      },
      {
        "$group": {
          "_id": {
            "category": "$predicted_category",
            "location": "$location"
          },
          "count": {
            "$sum": 1
          },
          "open_count": {
            "$sum": {
              "$cond": [
                {
                  "$in": [
                    "$status",

                      "Pending",
                      "In Progress"

                  ]
                },
                1,
                0
              ]
            }
          },
          "first_reported": {
            "$min": "$created_at"

            "$max": "$created_at"

          }
        }
      },
      {
        "$match": {
          "count": {

--- 8. feedback_by_department  (feedback collection)  -  Average student rating per department by JOINING feedback with complaints.

        }
      },
      {
        "$unwind": "$complaint"
      },

        "$group": {

            "$sum": 1

--- 11. resolution_time_buckets  (complaints collection)  -  Resolved complaints grouped into time buckets (<1 day, 1-3 days, 3-7 days, 7+ days).   ($match + $bucket)
    pipeline:
    [
      {
        "$match": {
          "status": "Resolved",
          "resolved_at": {
            "$ne": null
          }
        }
      },
      {
        "$addFields": {
          "hours": {
            "$divide": [
              {
                "$subtract": [
                  "$resolved_at",
                  "$created_at"
                ]
              },
              3600000
            ]
          }
        }
      },
      {
        "$bucket": {
          "groupBy": "$hours",
          "boundaries": [
            0,
            24,
            72,
            168
          ],
          "default": "168+",
          "output": {
            "count": {
              "$sum": 1
            }
          }
        }
      }
    ]
    output:
    {'_id': 0, 'count': 29}
    {'_id': 24, 'count': 45}
    {'_id': 72, 'count': 32}
    {'_id': '168+', 'count': 3}
