--- comparison + logical operators ($in, $gte, $or)
    mongosh : db.complaints.find({$or:[{final_urgency:"High"},{predicted_category:{$in:["Security","Electricity"]}}], created_at:{$gte: ISODate(<7 days ago>)}})

    mongosh : db.complaints.find({title:{$regex:"^DEMO", $options:"i"}}).count()

==============================================================================
3. UPDATE
==============================================================================

--- update_one - $set the department and $push a status-history entry
    mongosh : db.complaints.updateOne({complaint_id:"CMP1156"}, {$set:{status:"In Progress", assigned_department:"IT Department"}, $push:{status_history:{...}}})
    {'status': 'In Progress',
      'assigned_department': 'IT Department',
      'status_history': [{'status': 'Pending'}, {'status': 'In Progress'}]}

--- update_many - assign every pending DEMO complaint to a department
    mongosh : db.complaints.updateMany({title:/^DEMO/, status:"Pending"}, {$set:{assigned_department:"General Administration"}})
    {'matched': 3, 'modified': 3}

--- find_one_and_update - atomically change and return the new document (admin urgency override)
    mongosh : db.complaints.findOneAndUpdate({complaint_id:"CMP1156"}, {$set:{final_urgency:"High", urgency_overridden:true}}, {returnDocument:"after"})
    {'complaint_id': 'CMP1156',
      'predicted_urgency': 'Medium',
      'final_urgency': 'High',
      'urgency_overridden': True}

--- upsert - create the department document if it does not exist
    mongosh : db.departments.updateOne({department_name:"Demo Dept"}, {$set:{...}}, {upsert:true})
    {'upserted_id': '6af9abc57c22f2a0c4564ae6'}

--- Auto-increment ids: the counters document is changed with $inc + upsert inside next_id()
    mongosh : db.counters.findOneAndUpdate({_id:"complaints"}, {$inc:{seq:1}}, {upsert:true, returnDocument:"after"})
    {'_id': 'complaints', 'seq': 159}
